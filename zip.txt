import subprocess
import os

zip_path = "/home/Django/80017209-20260420.zip"
extract_to = "/home/Django/extracted_files"
txt_file = "80017209.txt"

# 解压 ZIP 文件
os.makedirs(extract_to, exist_ok=True)
try:
    subprocess.run(["unzip", zip_path, "-d", extract_to], check=True)
    print("解压成功")
except subprocess.CalledProcessError as e:
    print(f"解压失败: {e}")
    exit(1)
except FileNotFoundError:
    print("错误: 未找到 unzip 命令，请确保已安装 unzip")
    exit(1)

# 读取文件（尝试 GBK 和 UTF-8 编码）
file_path = os.path.join(extract_to, txt_file)
if not os.path.exists(file_path):
    print(f"错误: 文件 {txt_file} 不存在")
    exit(1)

encodings = ["gbk", "utf-8", "gb18030"]  # 常见中文编码
for encoding in encodings:
    try:
        with open(file_path, "r", encoding=encoding) as f:
            print(f"成功以 {encoding} 编码读取文件:")
            print(f.read())
        break
    except UnicodeDecodeError:
        continue
else:
    print("错误: 无法用常见编码（GBK/UTF-8/GB18030）解码文件")
